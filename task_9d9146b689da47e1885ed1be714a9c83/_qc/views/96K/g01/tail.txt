527:     @staticmethod
528:     def get_use_ops(response_op):
529:         """
530:         Starting from ResponseOp, recursively traverse the front OPs. Getting
531:         all used ops and the post op list of each op (excluding ResponseOp)
532: 
533:         Args:
534:             response_op: ResponseOp
535: 
536:         Returns:
537:             used_ops: used ops, set
538:             succ_ops_of_use_op: op and the next op list, dict.
539:             
540:         """
541:         unique_names = set()
542:         used_ops = set()
543:         succ_ops_of_use_op = {}  # {op_name: succ_ops}
544:         que = Queue.Queue()
545:         que.put(response_op)
546:         while que.qsize() != 0:
547:             op = que.get()
548:             for pred_op in op.get_input_ops():
549:                 if pred_op.name not in succ_ops_of_use_op:
550:                     succ_ops_of_use_op[pred_op.name] = []
551:                 if op != response_op:
552:                     succ_ops_of_use_op[pred_op.name].append(op)
553:                 if pred_op not in used_ops:
554:                     que.put(pred_op)
555:                     used_ops.add(pred_op)
556:                     # check the name of op is globally unique
557:                     if pred_op.name in unique_names:
558:                         _LOGGER.critical("Failed to get used Ops: the"
559:                                          " name of Op must be unique: {}".
560:                                          format(pred_op.name))
561:                         os._exit(-1)
562:                     unique_names.add(pred_op.name)
563:         return used_ops, succ_ops_of_use_op
