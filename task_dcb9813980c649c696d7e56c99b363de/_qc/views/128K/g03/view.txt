```java
private Double getBoundAsDouble(RexWindowBound bound)
{
  if (bound.isUnbounded()) return null;
  if (bound.isCurrentRow()) return 0.0d;
  final RexLiteral constant = window.constants.get(((RexInputRef) bound.getOffset()).getIndex() - sig.size());

}
```

```java
public WindowFrame getWindowFrame()
{
  if (group.lowerBound.isUnbounded() && group.upperBound.isUnbounded()) {
    return WindowFrame.unbounded();
  }
  if (group.isRows) {
    return WindowFrame.rows(getBoundAsInteger(group.lowerBound), getBoundAsInteger(group.upperBound));
  }
  if (hasOffsetBound(group.lowerBound) || hasOffsetBound(group.upperBound)) {
    // RANGE with at least one finite offset
    // validate single ordering key + type via sig
    return WindowFrame.range(getBoundAsDouble(group.lowerBound), getBoundAsDouble(group.upperBound), getSingleOrderingColumn());
  }
  return WindowFrame.groups(getBoundAsInteger(group.lowerBound), getBoundAsInteger(group.upperBound), getOrderingColumNames());
}
```
